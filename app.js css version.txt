import React, { useEffect, useRef, useState } from "react";

function App() {
  /* =========================================================
     STATE
  ========================================================= */

  const [darkMode, setDarkMode] = useState(() => {
    try {
      return localStorage.getItem("theme") === "dark";
    } catch {
      return false;
    }
  });

  const [loading, setLoading] = useState(true);
  const [activeCampaign, setActiveCampaign] = useState(0);
  const [activeEditorial, setActiveEditorial] = useState(0);
  const [activeImageIndex, setActiveImageIndex] = useState(0);
  const [lightboxImage, setLightboxImage] = useState(null);
  const [mobileMenu, setMobileMenu] = useState(false);

  // Touch handlers for mobile swipe
  const [touchStart, setTouchStart] = useState(null);
  const [touchEnd, setTouchEnd] = useState(null);

  /* =========================================================
     DATA
  ========================================================= */

  const marqueeWords = [
    "EDITORIAL",
    "RUNWAY",
    "PORTRAIT",
    "FASHION",
    "CAMPAIGN",
    "CAIRO",
  ];

  const editorials = [
    {
      title: "Urban Chic",
      images: [
        "/photo1.png",
        "/photo6.png",
      ],
    },
    {
      title: "Avant-Garde",
      images: [
        "/photo2.png",
        "/photo6.png",
      ],
    },
    {
      title: "Studio Portrait",
      images: [
        "/photo7.png",
        "/photo5.png",
      ],
    },
  ];

  const campaigns = [
    {
      title: "Summer Editorial",
      image: "/photo5.png",
    },
    {
      title: "Editorial Portrait",
      image: "/photo6.png",
    },
    {
      title: "Formal Series",
      image: "/photo1.png",
    },
    {
      title: "Streetwear",
      image: "/photo2.png",
    },
  ];

  // Selected Works: Block 2 is the quote "Fashion is about dreaming"
  const selectedWorks = [
    { type: "image", src: "/photo5.png", alt: "Work 1" },
    { type: "quote-dream", quote: "Fashion is about dreaming.", author: "— SELECTED WORKS" },
    { type: "image", src: "/photo6.png", alt: "Work 2" },
    { type: "image", src: "/photo1.png", alt: "Work 3" },
    { type: "image", src: "/photo2.png", alt: "Work 4" },
    { type: "quote-style", quote: "Style is identity.", author: "— M.NASR" },
    { type: "image", src: "/photo7.png", alt: "Work 5" },
  ];

  const photography = [
    "/lens1.jpeg",
    "/lens2.jpeg",
    "/lens3.jpeg",
  ];

  /* =========================================================
     REFS & THEME
  ========================================================= */

  const navRef = useRef(null);

  useEffect(() => {
    document.body.className = darkMode ? "dark-mode" : "light-mode";

    try {
      localStorage.setItem("theme", darkMode ? "dark" : "light");
    } catch {}
  }, [darkMode]);

  /* =========================================================
     LOADER & SCROLL REVEAL
  ========================================================= */

  useEffect(() => {
    const timer = setTimeout(() => {
      setLoading(false);
    }, 1400);

    return () => clearTimeout(timer);
  }, []);

  useEffect(() => {
    const elements = document.querySelectorAll(".reveal");

    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            entry.target.classList.add("visible");
            observer.unobserve(entry.target);
          }
        });
      },
      { threshold: 0.12 }
    );

    elements.forEach((el) => observer.observe(el));

    return () => observer.disconnect();
  }, [loading]);

  /* =========================================================
     NAVIGATION
  ========================================================= */

  const scrollToSection = (id) => {
    setMobileMenu(false);
    const section = document.getElementById(id);
    if (section) {
      section.scrollIntoView({ behavior: "smooth" });
    }
  };

  /* =========================================================
     EDITORIAL SLIDER & COLLECTION SWAPPING
  ========================================================= */

  const nextEditorial = () => {
    setActiveImageIndex(0);
    setActiveEditorial((prev) => (prev + 1) % editorials.length);
  };

  const prevEditorial = () => {
    setActiveImageIndex(0);
    setActiveEditorial(
      (prev) => (prev - 1 + editorials.length) % editorials.length
    );
  };

  const nextImage = (e) => {
    e?.stopPropagation();
    const currentImages = editorials[activeEditorial].images;
    setActiveImageIndex((prev) => (prev + 1) % currentImages.length);
  };

  const prevImage = (e) => {
    e?.stopPropagation();
    const currentImages = editorials[activeEditorial].images;
    setActiveImageIndex(
      (prev) => (prev - 1 + currentImages.length) % currentImages.length
    );
  };

  // Mobile Swipe Handlers
  const handleTouchStart = (e) => {
    setTouchEnd(null);
    setTouchStart(e.targetTouches[0].clientX);
  };

  const handleTouchMove = (e) => {
    setTouchEnd(e.targetTouches[0].clientX);
  };

  const handleTouchEnd = () => {
    if (!touchStart || !touchEnd) return;
    const distance = touchStart - touchEnd;
    if (distance > 50) nextEditorial();
    else if (distance < -50) prevEditorial();
  };

  /* =========================================================
     LOADING SCREEN
  ========================================================= */

  if (loading) {
    return (
      <>
        <style>{styles}</style>
        <div className="loader">
          <div className="loader-inner">
            <span>MOHAMMED</span>
            <span>NASR</span>
          </div>
          <div className="loader-line">
            <div />
          </div>
        </div>
      </>
    );
  }

  /* =========================================================
     CALCULATE ACTIVE & SIDE COLLECTIONS
  ========================================================= */

  const currentEditorial = editorials[activeEditorial];
  const activeImgSrc = currentEditorial.images[activeImageIndex] || currentEditorial.images[0];

  const sideIndex1 = (activeEditorial + 1) % editorials.length;
  const sideIndex2 = (activeEditorial + 2) % editorials.length;

  return (
    <div className="app">
      <style>{styles}</style>

      {/* =====================================================
          NAVBAR
      ===================================================== */}
      <nav ref={navRef} className={`navbar ${mobileMenu ? "menu-open" : ""}`}>
        <div className="logo" onClick={() => scrollToSection("home")}>
          M.NASR
        </div>

        <div className={`nav-links ${mobileMenu ? "show" : ""}`}>
          {[
            ["Home", "home"],
            ["About", "about"],
            ["Stats", "stats"],
            ["Editorials", "portfolio"],
            ["Campaigns", "campaigns"],
            ["Works", "works"],
            ["Lens", "photography"],
            ["Reel", "reel"],
            ["Contact", "contact"],
          ].map(([label, id]) => (
            <button key={id} onClick={() => scrollToSection(id)}>
              {label}
            </button>
          ))}
        </div>

        <div className="nav-actions">
          <button
            className="theme-btn"
            onClick={() => setDarkMode(!darkMode)}
            aria-label="Toggle theme"
          >
            {darkMode ? "☀" : "◐"}
          </button>

          <button
            className="menu-btn"
            onClick={() => setMobileMenu(!mobileMenu)}
          >
            {mobileMenu ? "×" : "☰"}
          </button>
        </div>
      </nav>

      {/* =====================================================
          HERO
      ===================================================== */}
      <header id="home" className="hero">
        <div className="hero-bg-text">MODEL</div>

        <div className="hero-content reveal">
          <div className="hero-image-wrapper">
            <div className="hero-image-border" />
            <img
              src="/profile.png"
              alt="Mohammed Nasr"
              className="hero-image"
            />
          </div>

          <div className="hero-kicker">CAIRO · EGYPT</div>

          <h1>
            Mohammed
            <br />
            <span>Nasr</span>
          </h1>

          <p>Editorial & Runway Model</p>

          <button
            className="scroll-cue"
            onClick={() => scrollToSection("about")}
          >
            <span>SCROLL TO EXPLORE</span>
            <div className="scroll-line" />
          </button>
        </div>

        <div className="hero-side-text">
          <span>01</span>
          <span>PORTFOLIO</span>
        </div>
      </header>

      {/* =====================================================
          MARQUEE (Seamless 100% Loop)
      ===================================================== */}
      <section className="marquee">
        <div className="marquee-track">
          {[1, 2].map((block) => (
            <div className="marquee-content" key={block}>
              {[1, 2, 3].map((rep) =>
                marqueeWords.map((word, idx) => (
                  <React.Fragment key={`${rep}-${idx}`}>
                    <span>{word}</span>
                    <i>✦</i>
                  </React.Fragment>
                ))
              )}
            </div>
          ))}
        </div>
      </section>

      {/* =====================================================
          ABOUT
      ===================================================== */}
      <section id="about" className="section about-section">
        <div className="section-number">01 / ABOUT</div>

        <div className="about-grid">
          <div className="about-image reveal">
            <img src="/photo5.png" alt="Mohammed Nasr portrait" />
            <div className="image-caption">
              MOHAMMED NASR
              <span>EDITORIAL PORTRAIT</span>
            </div>
          </div>

          <div className="about-content reveal">
            <div className="small-label">STATEMENT</div>

            <h2>
              Presence before
              <br />
              <em>performance.</em>
            </h2>

            <p>
              Editorial modeling is less about filling the frame and more about
              controlling it. Every movement, expression and pause has a purpose.
            </p>

            <p>
              My approach is built around stillness, structure and visual
              awareness — allowing the photographer, styling and lighting to work
              together.
            </p>

            <div className="about-signature">M.N</div>
          </div>
        </div>
      </section>

      {/* =====================================================
          STATS
      ===================================================== */}
      <section id="stats" className="section stats-section">
        <div className="section-number">02 / VITALS</div>

        <div className="stats-heading reveal">
          <div>
            <div className="small-label">SPECIFICATIONS</div>
            <h2>
              Vital
              <br />
              Statistics
            </h2>
          </div>

          <p className="editorial-desc">
            Measurements & details for
            <br />
            casting and editorial work.
          </p>
        </div>

        <div className="stats-list reveal">
          {[
            ["01", "Height", "1.86 m"],
            ["02", "Weight", "78 kg"],
            ["03", "Chest / Waist", "98 / 80 cm"],
            ["04", "Shoe Size", "EU 43"],
            ["05", "Hair", "Black"],
            ["06", "Eyes", "Dark Brown"],
            ["07", "Based In", "Cairo, Egypt"],
            ["08", "Languages", "Arabic · English"],
          ].map(([num, label, value]) => (
            <div className="stat-row" key={num}>
              <span className="stat-num">{num}</span>
              <span className="stat-label">{label}</span>
              <span className="stat-value">{value}</span>
            </div>
          ))}
        </div>
      </section>

      {/* =====================================================
          EDITORIALS (Desktop 3-Block Swappable View)
      ===================================================== */}
      <section id="portfolio" className="section editorial-section">
        <div className="section-number">03 / EDITORIALS</div>

        <div className="editorial-header reveal">
          <div>
            <div className="small-label">SELECTED WORK</div>
            <h2>
              Editorial
              <br />
              <em>Stories.</em>
            </h2>
          </div>

          <div className="editorial-counter">
            <span>{String(activeEditorial + 1).padStart(2, "0")}</span>
            <div />
            <span>{String(editorials.length).padStart(2, "0")}</span>
          </div>
        </div>

        <div
          className="editorial-slider reveal"
          onTouchStart={handleTouchStart}
          onTouchMove={handleTouchMove}
          onTouchEnd={handleTouchEnd}
        >
          {/* Main Large Block (Active Collection) */}
          <div className="editorial-main-image">
            <button
              className="slider-arrow left"
              onClick={prevImage}
              title="Previous photo in this collection"
              aria-label="Previous photo"
            >
              ←
            </button>

            <img
              key={`${activeEditorial}-${activeImageIndex}`}
              src={activeImgSrc}
              alt={`${currentEditorial.title} frame ${activeImageIndex + 1}`}
              onClick={() => setLightboxImage(activeImgSrc)}
            />

            <div className="editorial-overlay">
              <span>
                ACTIVE COLLECTION · PHOTO {activeImageIndex + 1} / {currentEditorial.images.length}
              </span>
              <h3>{currentEditorial.title}</h3>
            </div>

            <button
              className="slider-arrow right"
              onClick={nextImage}
              title="Next photo in this collection"
              aria-label="Next photo"
            >
              →
            </button>
          </div>

          {/* Desktop Right Side (2 Other Collections) */}
          <div className="editorial-side">
            {[sideIndex1, sideIndex2].map((idx) => {
              const col = editorials[idx];
              return (
                <div
                  key={col.title}
                  className="editorial-side-card"
                  onClick={() => {
                    setActiveImageIndex(0);
                    setActiveEditorial(idx);
                  }}
                  title={`Click to view ${col.title}`}
                >
                  <img src={col.images[0]} alt={col.title} />
                  <div className="side-card-overlay">
                    <span className="side-card-tag">COLLECTION</span>
                    <h4>{col.title}</h4>
                    <span className="side-card-action">SWAP ↗</span>
                  </div>
                </div>
              );
            })}
          </div>
        </div>

        {/* Outer Block Collection Navigation */}
        <div className="editorial-controls">
          <button onClick={prevEditorial}>PREVIOUS COLLECTION</button>

          <div className="editorial-dots">
            {editorials.map((_, index) => (
              <button
                key={index}
                className={index === activeEditorial ? "active" : ""}
                onClick={() => {
                  setActiveImageIndex(0);
                  setActiveEditorial(index);
                }}
                aria-label={`Go to collection ${index + 1}`}
              />
            ))}
          </div>

          <button onClick={nextEditorial}>NEXT COLLECTION</button>
        </div>
      </section>

      {/* =====================================================
          CAMPAIGNS
      ===================================================== */}
      <section id="campaigns" className="section campaigns-section">
        <div className="section-number">04 / CAMPAIGNS</div>

        <div className="campaign-heading reveal">
          <div className="small-label">FEATURED SERIES</div>
          <h2>
            Campaign
            <br />
            <em>Direction.</em>
          </h2>
        </div>

        <div className="campaign-layout reveal">
          <div className="campaign-panels">
            {campaigns.map((campaign, index) => (
              <div
                key={campaign.title}
                className={`campaign-panel ${
                  activeCampaign === index ? "active" : ""
                }`}
                onMouseEnter={() => setActiveCampaign(index)}
                onClick={() => setActiveCampaign(index)}
              >
                <img src={campaign.image} alt={campaign.title} />
                <div className="campaign-number">0{index + 1}</div>
                <h3>{campaign.title}</h3>
                <span className="campaign-plus">+</span>
              </div>
            ))}
          </div>

          <div className="campaign-description">
            <div className="small-label">
              {String(activeCampaign + 1).padStart(2, "0")} / SERIES
            </div>

            <blockquote>
              “Fashion is not simply
              <br />
              clothing. It is
              <em> visual language.</em>”
            </blockquote>

            <p className="editorial-desc">
              A selection of visual concepts built around silhouette, composition,
              styling and controlled expression.
            </p>
          </div>
        </div>
      </section>

      {/* =====================================================
          SELECTED WORKS
      ===================================================== */}
      <section id="works" className="section works-section">
        <div className="section-number">05 / SELECTED WORKS</div>

        <div className="works-header reveal">
          <div>
            <div className="small-label">PORTFOLIO</div>
            <h2>
              Selected
              <br />
              <em>Works.</em>
            </h2>
          </div>

          <p className="editorial-desc">
            A visual collection of
            <br />
            recent editorial work.
          </p>
        </div>

        <div className="masonry reveal">
          {selectedWorks.map((item, index) => {
            if (item.type === "quote-dream") {
              return (
                <div className="quote-card quote-dream" key="quote-dream">
                  <span>“</span>
                  <p>{item.quote}</p>
                  <small>{item.author}</small>
                </div>
              );
            }

            if (item.type === "quote-style") {
              return (
                <div className="quote-card quote-style" key="quote-style">
                  <span>“</span>
                  <p>{item.quote}</p>
                  <small>{item.author}</small>
                </div>
              );
            }

            return (
              <div
                className="work-image"
                key={item.src + index}
                onClick={() => setLightboxImage(item.src)}
              >
                <img src={item.src} alt={item.alt} />
                <span>VIEW +</span>
              </div>
            );
          })}
        </div>
      </section>

      {/* =====================================================
          PHOTOGRAPHY (LENS)
      ===================================================== */}
      <section id="photography" className="section photography-section">
        <div className="section-number">06 / LENS</div>

        <div className="lens-header reveal">
          <div>
            <div className="small-label">PERSONAL PROJECT</div>
            <h2>
              Behind
              <br />
              <em>the Lens.</em>
            </h2>
          </div>

          <p className="editorial-desc">
            Landscape photography
            <br />
            captured by me.
          </p>
        </div>

        <div className="lens-grid reveal">
          {photography.map((image, index) => (
            <div
              className={`lens-image lens-${index + 1}`}
              key={image}
              onClick={() => setLightboxImage(image)}
            >
              <img src={image} alt={`Landscape ${index + 1}`} />
              <div className="lens-label">0{index + 1}</div>
              <div className="lens-view">VIEW IMAGE</div>
            </div>
          ))}
        </div>
      </section>

      {/* =====================================================
          SHOWREEL
      ===================================================== */}
      <section id="reel" className="section reel-section">
        <div className="section-number">07 / SHOWREEL</div>

        <div className="reel-header reveal">
          <div>
            <div className="small-label">MOTION</div>
            <h2>
              Show
              <br />
              <em>Reel.</em>
            </h2>
          </div>

          <span>2026</span>
        </div>

        <div className="video-wrapper reveal">
          <video
            src="/video.mp4"
            autoPlay
            loop
            muted
            playsInline
          />

          <div className="video-overlay">
            <span>MOHAMMED NASR</span>
            <span>EDITORIAL / 2026</span>
          </div>
        </div>
      </section>

      {/* =====================================================
          CONTACT
      ===================================================== */}
      <section id="contact" className="contact-section">
        <div className="contact-inner">
          <div className="contact-top reveal">
            <span>08 / CONTACT</span>
            <span>CAIRO · EGYPT</span>
          </div>

          <div className="contact-content reveal">
            <div className="small-label">AVAILABLE FOR</div>

            <h2>
              Let's make
              <br />
              something
              <br />
              <em>visual.</em>
            </h2>

            <p className="editorial-desc">
              For bookings, casting inquiries, collaborations and editorial work.
            </p>
          </div>

          <div className="contact-links reveal">
            <a
              href="https://www.instagram.com/mo7ammed__medoo"
              target="_blank"
              rel="noopener noreferrer"
            >
              <span>Instagram</span>
              <span>@mo7ammed__medoo ↗</span>
            </a>

            <a href="mailto:mohammednasrsmail@gmail.com">
              <span>Email</span>
              <span>mohammednasrsmail@gmail.com ↗</span>
            </a>

            <a
              href="https://wa.me/201156108363"
              target="_blank"
              rel="noopener noreferrer"
            >
              <span>WhatsApp</span>
              <span>+20 115 610 8363 ↗</span>
            </a>
          </div>
        </div>
      </section>

      {/* =====================================================
          FOOTER
      ===================================================== */}
      <footer>
        <div className="footer-logo">M.NASR</div>
        <p>Editorial & Runway Model</p>
        <span>© {new Date().getFullYear()} Mohammed Nasr</span>
      </footer>

      {/* =====================================================
          LIGHTBOX
      ===================================================== */}
      {lightboxImage && (
        <div className="lightbox" onClick={() => setLightboxImage(null)}>
          <button
            className="lightbox-close"
            onClick={() => setLightboxImage(null)}
          >
            ×
          </button>
          <img
            src={lightboxImage}
            alt="Expanded portfolio"
            onClick={(e) => e.stopPropagation()}
          />
        </div>
      )}
    </div>
  );
}
